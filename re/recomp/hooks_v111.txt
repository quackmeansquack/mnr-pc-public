# Guest functions replaced by host implementations (in host.c), for the v1.11 patch EBOOT (bin_v111/EBOOT.elf).
# Addresses mapped from hooks_v100.txt (disc EBOOT) with re/scripts/map_addrs.py. Format: <hex address> <host function>
# The translator emits f_<address> as a call to the host function instead of the guest body.

# qTaskFunctionManager::OnSpuElfLoaded - binds a qTask function's SPU entry when its SPU program is
# registered. Reimplemented on the host (same binding); the bound tasks then run their SPU code through
# prehook_qtask_run (several PPU implementations are empty or incompatible stubs).
1e4130 hook_qtask_on_spu_elf_loaded

# Debug pre-hooks (host function runs first, then the original guest body):
# lua_getfield / lua_gettable record (key, index, caller) per thread; the Lua panic handler dumps them.

113f8 prehook_lua_gettable pre
14c18 prehook_lua_panic pre

# qTask SPU launcher: (table, slot, spu job, task, ppu queue, manager, 0). Tasks that carry an SPU job
# would start the SPU runner program (VA 0xF18580); queue them for the PPU workers instead.
7cc470 hook_qtask_spu_launch

# qTask runner: runs bound tasks' SPU functions on the SPU core; task statistics (MNR_QTASK_STATS=1)
7c698c prehook_qtask_run pre

# Havok SPU thread pool waitForCompletion (0xAEC180): waits once per SPU worker on a semaphore the SPU
# tasks release. The SPU workers never run here and the calling thread has already processed the job
# queue itself, so just clear the "waiting" flag.
b04fc8 hook_havok_spu_wait

# Asynchronous Havok physics job (0x482F8 adds it to the thread pool, 0x48468 waits for it): only the
# SPU pool would process it. The byte at 0xECB74C enables that path; force it off before both.
1c5678 prehook_async_physics_off pre
1c57e8 prehook_async_physics_off pre
